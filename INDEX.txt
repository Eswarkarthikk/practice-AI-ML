export default function MyScreen() {
  const { transactions, addTransaction } = useTransactions();

  const handleAdd = async () => {
    await addTransaction({
      amount: 50,
      date: '2026-02-19',
      category: 'food',
      type: 'expense',
      note: 'Lunch',
      isRecurring: true,              // ← NEW: Mark as recurring
      recurrenceType: 'monthly',      // ← NEW: Daily/Weekly/Monthly/Yearly
      lastGeneratedDate: '2026-02-19',
      isEditable: true,
    });
  };
}
