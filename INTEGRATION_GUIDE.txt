3️⃣ DASHBOARD (app/(tabs)/index.tsx) ADD HEALTH SCORE & BUDGET

```typescript
import { useTransactions } from '@/lib/features/transactions/TransactionContext';
import { useBudget } from '@/lib/features/budgets/BudgetContext';
import { useGoals } from '@/lib/features/goals/GoalContext';
import { calculateFinancialHealthScore } from '@/lib/features/analytics/advancedAnalytics';
import { ProgressBar } from '@/components/shared/ProgressBar';
import Colors from '@/lib/theme/colors';

export default function DashboardScreen() {
  const { transactions } = useTransactions();
  const { budgets, getBudgetStats } = useBudget();
  const { goals } = useGoals();

  // Get current month (format: YYYY-MM)
  const currentMonth = new Date().toISOString().substring(0, 7);

  // Financial health score
  const healthScore = calculateFinancialHealthScore(transactions, budgets, goals);
  
  // Budget stats
  const budgetStats = getBudgetStats(currentMonth);

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.greeting}>Welcome back! 👋</Text>
      
      {/* Balance Card */}
      <BalanceCard transactions={transactions} />

      {/* Health Score Card */}
      <Card padding={16}>
        <Text style={styles.sectionTitle}>Financial Health</Text>
        <Text style={styles.scoreText}>{healthScore.score}/100</Text>
        <Text style={styles.ratingText}>{healthScore.rating}</Text>
        <ProgressBar percentage={healthScore.score} showPercentage />
      </Card>

      {/* Budget Progress Card */}
      {budgetStats.totalLimit > 0 && (
        <Card padding={16}>
          <Text style={styles.sectionTitle}>Monthly Budget</Text>
          <Text style={styles.amountText}>
            ${budgetStats.totalSpent.toFixed(2)} / ${budgetStats.totalLimit.toFixed(2)}
          </Text>
          <ProgressBar percentage={budgetStats.percentageUsed} />
          <Text style={{
            color: budgetStats.status === 'safe' ? Colors.success :
                   budgetStats.status === 'warning' ? Colors.warning :
                   Colors.danger
          }}>
            {budgetStats.status.toUpperCase()}
          </Text>
        </Card>
      )}

      {/* Recent Transactions */}
      <RecentTransactions transactions={transactions.slice(0, 5)} />
    </ScrollView>
  );
}
```

```typescript
import { useTransactions } from '@/lib/features/transactions/TransactionContext';
```

```typescript
import { exportBackup, restoreBackup } from '@/lib/features/settings/backup';
import { Button } from '@/components/shared/Button';
import { Card } from '@/components/shared/Card';

export default function SettingsScreen() {
  const [appLockEnabled, setAppLockEnabled] = useState(false);
  const { transactions, loadTransactions } = useTransactions();

  useEffect(() => {
    checkLockStatus();
  }, []);

  const checkLockStatus = async () => {
    const enabled = await AppLock.isLockEnabled();
    setAppLockEnabled(enabled);
  };

  const handleToggleLock = async () => {
    if (appLockEnabled) {
      await AppLock.disableLock();
      setAppLockEnabled(false);
    } else {
      const enabled = await AppLock.enableLock(true);
      setAppLockEnabled(enabled);
    }
  };

  const handleExportBackup = async () => {
    try {
      await exportBackup();
      Alert.alert('Success', 'Backup exported successfully');
    } catch (error) {
      Alert.alert('Error', 'Failed to export backup');
    }
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>Settings</Text>

      <Text style={styles.sectionTitle}>Security</Text>
      <Card padding={16}>
        <View style={styles.settingRow}>
```

🔒 BONUS: LOCK SCREEN (app/lock.tsx)

Add at app root to protect data:

```typescript
import { useState, useEffect } from 'react';
import { View, Text, TextInput, Button } from 'react-native';
import { AppLock } from '@/lib/features/settings/appLock';
import { HapticFeedback } from '@/lib/features/settings/haptics';

export default function LockScreen({ onUnlock }: { onUnlock: () => void }) {
  const [pin, setPin] = useState('');
  const [biometricAvailable, setBiometricAvailable] = useState(false);

  useEffect(() => {
    checkBiometric();
    attemptBiometric();
  }, []);

  const checkBiometric = async () => {
    const available = await AppLock.isBiometricAvailable();
    setBiometricAvailable(available);
  };

  const attemptBiometric = async () => {
    const success = await AppLock.authenticateBiometric();
    if (success) {
      await HapticFeedback.success();
      onUnlock();
    }
  };

  const handlePINSubmit = async () => {
    const verified = await AppLock.verifyPIN(pin);
    if (verified) {
      await HapticFeedback.success();
      onUnlock();
    } else {
      await HapticFeedback.error();
      setPin('');
      alert('Incorrect PIN');
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>App Locked</Text>

      {biometricAvailable && (
        <Button title="Use Biometric" onPress={attemptBiometric} />
      )}

      <TextInput
        style={styles.pinInput}
        placeholder="Enter PIN"
        secureTextEntry
        value={pin}
        onChangeText={setPin}
      />
      <Button title="Unlock" onPress={handlePINSubmit} />
    </View>
  );
}
```
